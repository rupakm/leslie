\section{Proof Decomposition}

The end goal is the correctness of \textsf{ABA.Impl}.
We establish it via a simulation by \textsf{ABA.Spec} and transport the spec's correctness.
The simulation itself is decomposed recursively along the sub-protocol structure.

For a protocol $P$ with concrete sub-protocols $Q_1,\dots,Q_k$, the step ``\textsf{P.Spec}
simulates \textsf{P.Impl}'' is split into:
\begin{enumerate}
    \item (Substitution) $\mathsf{P.Impl}[\mathsf{Q_1.Spec},\dots,\mathsf{Q_k.Spec}]$ is simulated
    by \textsf{P.Impl}, using the child simulations \textsf{$Q_i$.Spec} simulates
    \textsf{$Q_i$.Impl} and compositionality.
    \item (Core) \textsf{P.Spec} simulates the hybrid
    $\mathsf{P.Impl}[\mathsf{Q_1.Spec},\dots,\mathsf{Q_k.Spec}]$.
    \item (Transitivity) Compose (1) and (2).
\end{enumerate}
Leaves of the recursion (no sub-protocol, or only the black-box \textsf{AVSS.Spec}) skip the
substitution step.

\subsection{Top-level}

\begin{theorem}\label{thm:ABACorrect}
    \lean{ABA.Correct}
    %\leanok
    \uses{ABA:Impl, ABA:Spec}
    \textsf{ABA.Impl} satisfies Validity, Agreement and Almost-Sure Termination.
\end{theorem}
\begin{proof}
    \uses{thm:ABASim,thm:Segala}
    According to Theorem \ref{thm:ABASim}, \textsf{ABA.Impl} $\sqsubseteq^f$
    \textsf{ABA.Spec}. Moreover, one can prove that \textsf{ABA.Spec} satisfies
    all three properties which can all be expressed as trace properties or fair trace
    distribution properties. Thus, by Theorem \ref{thm:Segala}, \textsf{ABA.Impl} satisfies all
    three properties.
\end{proof}

\subsection{ABA}

\begin{theorem}\label{thm:ABASim}
    \lean{ABA.Sim}
    %\leanok
    \uses{ABA:Impl, ABA:Spec}
    \textsf{ABA.Impl} $\sqsubseteq^f$ \textsf{ABA.Spec}.
\end{theorem}
\begin{proof}
    \uses{lem:ABASubst, lem:ABACore}
    Trivial by transitivity of $\sqsubseteq^f$ and Lemma \ref{lem:ABACore} and \ref{lem:ABASubst}.
\end{proof}

\begin{lemma}\label{lem:ABASubst}
    \lean{ABA.SubstSim}
    %\leanok
    \uses{ABA:Impl, ABA:Hybrid}
    \textsf{ABA.Impl} $\sqsubseteq^f$ \textsf{ABA.Hybrid}.
\end{lemma}
\begin{proof}
    \uses{thm:GBCASim, thm:WCCSim, lem:precong}
    Before abstraction, both systems only differ by replacing \textsf{GBCA.Impl} (resp.
    \textsf{WCC.Impl}) by \textsf{GBCA.Spec} (resp. \textsf{WCC.Spec}). However,
    \textsf{ABA.Impl} is partial-information so we need to relate its belief PLTS.
    According to Theorem \ref{thm:partialinfocompo}, the belief PLTS of the composition
    is isomorphic to the composition of the belief PLTS.
    Therefore, according to Theorem \ref{thm:GBCASim} and \ref{thm:WCCSim},
    Lemma \ref{lem:precong} followed by Lemma \ref{lem:abstract} imply \textsf{ABA.Impl}
    $\sqsubseteq^f$ \textsf{ABA.Hybrid}.
\end{proof}

\begin{lemma}\label{lem:ABACore}
    \lean{ABA.CoreSim}
    %\leanok
    \uses{ABA:Hybrid, ABA:Spec}
    \textsf{ABA.Hybrid} $\sqsubseteq^f$ \textsf{ABA.Spec}.
\end{lemma}
\begin{proof}
    \uses{thm:Segala}
    Both systems are full-information and probabilistic, thus, according to
    Theorem \ref{thm:Segala}, it suffices to build a progressive probabilistic
    forward simulation of \textsf{ABA.Hybrid} by \textsf{ABA.Spec}.
