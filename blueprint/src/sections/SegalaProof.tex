\section{Proof of Theorem \ref{thm:Segala}}
\label{app:ProofSegala}

In order to extend Segala's result to liveness, we need to display how a scheduler of $\pcal$ can
be translated into a scheduler of $\pcal'$. Unfortunately, Segala's proof does not follow that 
route \cite{Segala1995PhD}. Indeed, this construction is not trivial at all. Thus, we are going to
split it into multiple steps corresponding to the three main difficulties: weak transitions, 
simulation by probability distribution and translating $\pcal$ to $\pcal'$.

\subsection{Weak Transitions}

\begin{definition}[Weak Closure]
    The weak closure of a PLTS $\pcal$ is a PLTS, denoted $\pcal^w$, where transitions are replaced
    by weak transitions.
\end{definition}

\begin{lemma}
    \label{lem:weakclosuresound}
    Let $\pcal$ be a PLTS, $\text{tdist}(\pcal) = \text{tdist}(\pcal^w)$.
\end{lemma}
\begin{proof}
    The inclusion $\subseteq$ is trivial because any transition of $\pcal$ is a weak transition so
    schedulers of $\pcal$ are schedulers of $\pcal'$. However, the other inclusion $\supseteq$ is
    more tricky. [Done in Lean]
\end{proof}

\subsection{Distribution Monad}

\begin{definition}[Distribution Lifting]
    The distribution lifting of a PLTS $\pcal$ is the PLTS defined as 
    $\dcal(\pcal) := (\dcal(Q), \delta_{q_\star}, L, \dcal(T))$ where
    \[(\mu,l,\mathfrak{m}) \in \dcal(T) \iff \exists \mu' \in \dcal(Q), (\mu,l,\mu') \in H \wedge \sum_{\mu}\mathfrak{m}(\mu)\cdot\mu = \mu'\]
\end{definition}

\begin{lemma}
    \label{lem:distliftingsound}
    Let $\pcal$ be a PLTS, $\text{tdist}(\pcal) = \text{tdist}(\dcal(\pcal))$.
\end{lemma}
\begin{proof}
    The inclusion $\subseteq$ is trivial because Dirac distributions allow to encode any scheduler
    on $\pcal$ as a scheduler on $\pcal'$. However, the other inclusion $\supseteq$ is more tricky.
    [Done in Lean]
\end{proof}

\subsection{Strong Probabilistic Simulation}

\begin{definition}[Strong Probabilistic Simulation]
    A strong probabilistic simulation of $\pcal$ by $\pcal'$ is a relation $\scal \subseteq Q 
    \times Q'$ such that $(q_\star,q'_\star) \in \scal$ and, for all $(q,q') \in \scal$ and 
    $(q,l,\mu) \in T$, there exists $(q',l,\mu') \in T'$ such that $(\mu,\mu') \in \dcal(\scal)$.
\end{definition}

\begin{lemma}
    \label{lem:strongprobsim}
    Let $\pcal$ and $\pcal'$ be two PLTSs, there exists a probabilistic forward simulation of 
    $\pcal$ by $\pcal'$, if and only if, there exists a strong probabilistic simulation of $\pcal$
    by $\dcal(\pcal'^w)$.
\end{lemma}
\begin{proof}
    This proof is purely mechanical. It suffices to consider $\scal$ a probabilistic forward 
    simulation of $\pcal$ by $\pcal'$ (resp. a strong probabilistic simulation of $\pcal$ by 
    $\dcal(\pcal'^w)$) and check that it is a strong probabilistic simulation of $\pcal$ by 
    $\dcal(\pcal'^w)$ (resp. a probabilistic forward 
    simulation of $\pcal$ by $\pcal'$). [Done in Lean]
\end{proof}

\begin{lemma}
    \label{lem:strongprobsimsound}
    Let $\pcal$ and $\pcal'$ be two PLTSs, if there exists a strong probabilistic simulation of 
    $\pcal$ by $\pcal'$ then $\text{tdist}(\pcal) \subseteq \text{tdist}(\pcal')$.
\end{lemma}
\begin{proof}
    This is a simplified version of the original soundness proof which is tractable to formalize in 
    Lean. [Done in Lean]
\end{proof}

\begin{theorem}
    Let $\pcal$ and $\pcal'$ be two PLTSs, if there exists a probabilistic forward simulation of 
    $\pcal$ by $\pcal'$ then $\text{tdist}(\pcal) \subseteq \text{tdist}(\pcal')$.
\end{theorem}
\begin{proof}
    If there exists a probabilistic forward simulation of $\pcal$ by $\pcal'$ then, by Lemma 
    \ref{lem:strongprobsim}, there exists a strong probabilistic simulation of $\pcal$ by 
    $\dcal(\pcal'^w)$ so, by Lemma \ref{lem:strongprobsimsound}, \ref{lem:distliftingsound} and \ref{lem:weakclosuresound},
    \begin{equation*}
        \text{tdist}(\pcal) \subseteq \text{tdist}(\dcal(\pcal'^w)) = \text{tdist}(\pcal'^w) = \text{tdist}(\pcal')
    \end{equation*}
\end{proof}
