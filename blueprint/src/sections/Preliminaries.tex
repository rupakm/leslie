\section{Preliminaries}

This section gathers the standard material used in the rest of the document: probabilistic labelled transition systems, Segala-style relation lifting, weak transitions and probabilistic forward simulation, the specialisation to the non-probabilistic setting, partial-observation adversaries together with the belief construction, and CSP-style parallel composition together with its compositionality results.
Throughout, $\mathcal{D}(X)$ denotes the set of (discrete) probability distributions over a set $X$, and $\bar\mu \subseteq X$ denotes the support of $\mu \in \mathcal{D}(X)$.

\subsection{Probabilistic Labelled Transition Systems}

\begin{definition}[Probabilistic Labelled Transition System]\label{def:PLTS}
    %\lean{PLTS}
    %\leanok
    A \emph{probabilistic labelled transition system} (PLTS) is a tuple $\mathcal{P} = (Q, L, T, q_0)$ where
    \begin{itemize}
        \item $Q$ is a (countable) set of \emph{states};
        \item $L$ is a set of \emph{labels};
        \item $T \subseteq Q \times L \times \mathcal{D}(Q)$ is a set of \emph{probabilistic transitions};
        \item $q_0 \in Q$ is the \emph{initial state}.
    \end{itemize}
    A transition $(q, l, \mu) \in T$ is written $q \xrightarrow{l} \mu$ and is said to be \emph{enabled in $q$}.
    The label set is partitioned as $L = L_{\mathrm{int}} \sqcup L_{\mathrm{ext}}$ into \emph{internal} and \emph{external} labels.
\end{definition}

\begin{definition}[Execution and trace]\label{def:execution}
    \uses{def:PLTS}
    %\lean{PLTS.Execution}
    %\leanok
    An \emph{execution} of $\mathcal{P} = (Q, L, T, q_0)$ is a finite or infinite alternating sequence
    \[
        e \in (Q \cdot L)^* \cdot Q \;\cup\; (Q \cdot L)^\omega
    \]
    such that $e_0 = q_0$ and, whenever $2i+2 < |e|$, there exists $\mu \in \mathcal{D}(Q)$ with $(e_{2i}, e_{2i+1}, \mu) \in T$ and $e_{2i+2} \in \bar\mu$.
    The \emph{trace} of an execution $e$, written $\mathit{tr}(e)$, is the projection of $e$ onto $L_{\mathrm{ext}}$.
\end{definition}

\begin{definition}[Scheduler]\label{def:scheduler}
    \uses{def:PLTS, def:execution}
    %\lean{PLTS.Scheduler}
    %\leanok
    A \emph{scheduler} for $\mathcal{P}$ is a map $\sigma$ sending every finite execution $e$ to a sub-distribution over transitions whose support consists of transitions enabled in the last state of $e$.
\end{definition}

\begin{definition}[Probabilistic execution]\label{def:prob-execution}
    \uses{def:scheduler}
    %\lean{PLTS.ProbExecution}
    %\leanok
    Given a scheduler $\sigma$ and a starting state $q$, the \emph{probabilistic execution} $\mathbb{P}_{\sigma, q}$ is the unique measure on the $\sigma$-algebra generated by cones of executions characterised by
    \begin{itemize}
        \item $\mathbb{P}_{\sigma, q}(C_{q}) = 1$ and $\mathbb{P}_{\sigma, q}(C_{q'}) = 0$ for any $q' \ne q$;
        \item for $e' \cdot l \cdot q'$ a finite execution,
        \[
            \mathbb{P}_{\sigma, q}\!\left(C_{e' \cdot l \cdot q'}\right) \;=\; \mathbb{P}_{\sigma, q}(C_{e'}) \cdot \sigma(e')(t) \cdot \mu(q'),
        \]
        where $t = (\mathit{last}(e'), l, \mu)$ is the (necessarily unique) transition picked by $\sigma$ with label $l$.
    \end{itemize}
\end{definition}

\begin{definition}[Trace distribution]\label{def:trace-distribution}
    \uses{def:prob-execution, def:execution}
    %\lean{PLTS.TraceDistribution}
    %\leanok
    The trace function $\mathit{tr}$ is measurable from the $\sigma$-algebra of cones of executions to the $\sigma$-algebra of cones of traces. The \emph{trace distribution} induced by $\sigma$ from $q$ is the pushforward $\mathit{tr}_{*}\mathbb{P}_{\sigma, q}$.
\end{definition}

\begin{definition}[Trace distribution pre-order]\label{def:td-preorder}
    \uses{def:trace-distribution}
    %\lean{PLTS.TDPreorder}
    %\leanok
    Let $\mathcal{P}_1$ and $\mathcal{P}_2$ be two PLTS sharing the same set of external labels. We write $\mathcal{P}_1 \sqsubseteq_{\mathrm{TD}} \mathcal{P}_2$ when every trace distribution achievable by $\mathcal{P}_1$ from its initial state is also achievable by $\mathcal{P}_2$ from its initial state. This defines the \emph{trace distribution pre-order}.
\end{definition}

\subsection{Probabilistic Forward Simulation}

The trace distribution pre-order is a semantic object that is hard to manipulate directly. The standard tool to establish it is Segala's \emph{probabilistic forward simulation}, which lifts a state-to-distribution relation through the transition structure. We first need to lift relations between sets to relations between distributions.

\begin{definition}[Lifting of a relation between sets]\label{def:lifting-rel}
    %\lean{Relation.Lift}
    %\leanok
    Let $R \subseteq A \times B$. The \emph{lifting} of $R$ is the relation $R^\sharp \subseteq \mathcal{D}(A) \times \mathcal{D}(B)$ defined by $(\mu_A, \mu_B) \in R^\sharp$ iff there exists a weight function $w : A \times B \to [0,1]$ such that
    \begin{enumerate}
        \item $w(a, b) > 0 \Longrightarrow (a, b) \in R$;
        \item $\mu_A(a) = \sum_{b \in B} w(a, b)$ for every $a \in A$;
        \item $\mu_B(b) = \sum_{a \in A} w(a, b)$ for every $b \in B$.
    \end{enumerate}
\end{definition}

\begin{definition}[Lifting of a measure-valued relation]\label{def:lifting-dist}
    \uses{def:lifting-rel}
    %\lean{Relation.LiftDist}
    %\leanok
    Let $R \subseteq A \times \mathcal{D}(B)$. The \emph{lifting} $R^\sharp \subseteq \mathcal{D}(A) \times \mathcal{D}(B)$ is defined by $(\mu_A, \mu_B) \in R^\sharp$ iff there exists a family $(\mu_B^a)_{a \in \bar\mu_A}$ such that
    \begin{enumerate}
        \item $(a, \mu_B^a) \in R$ for every $a \in \bar\mu_A$;
        \item $\mu_B \;=\; \sum_{a \in A} \mu_A(a) \cdot \mu_B^a$.
    \end{enumerate}
\end{definition}

\begin{definition}[Hyper-transition]\label{def:hyper}
    \uses{def:lifting-rel, def:PLTS}
    %\lean{PLTS.Hyper}
    %\leanok
    For every label $l \in L$, the relation $\{(q, \mu) \mid (q, l, \mu) \in T\} \subseteq Q \times \mathcal{D}(Q)$ lifts (via Definition~\ref{def:lifting-dist}) to a relation $\xrightarrow{l} \subseteq \mathcal{D}(Q) \times \mathcal{D}(Q)$, called the \emph{hyper-transition} relation.
\end{definition}

\begin{definition}[Weak transitions]\label{def:weak-trans}
    \uses{def:scheduler, def:hyper}
    %\lean{PLTS.Weak}
    %\leanok
    A scheduler that schedules only transitions with internal labels and stops almost-surely defines a \emph{weak internal transition} $q \Rightarrow \mu$, where $\mu$ is the distribution on stopping states induced by the scheduler started in $q$.
    By Definition~\ref{def:lifting-dist} the relation $\Rightarrow$ extends to $\mathcal{D}(Q) \times \mathcal{D}(Q)$.
    For an external label $l \in L_{\mathrm{ext}}$, the \emph{weak external transition} $q \xRightarrow{l} \mu$ holds whenever there exist $\mu_1, \mu_2 \in \mathcal{D}(Q)$ with
    \[
        q \Rightarrow \mu_1 \xrightarrow{l} \mu_2 \Rightarrow \mu.
    \]
\end{definition}

\begin{definition}[Probabilistic forward simulation]\label{def:pfs}
    \uses{def:weak-trans, def:lifting-dist}
    %\lean{PLTS.ProbForwardSim}
    %\leanok
    A \emph{probabilistic forward simulation} of $\mathcal{P}_1$ by $\mathcal{P}_2$ is a relation $S \subseteq Q_1 \times \mathcal{D}(Q_2)$ such that $(q_0^1, \delta_{q_0^2}) \in S$ and, for every $(q_1, \mu_2) \in S$ and every transition $q_1 \xrightarrow{l} \mu_1$ of $\mathcal{P}_1$:
    \begin{enumerate}
        \item if $l$ is internal, there exists $\mu_2'$ with $\mu_2 \Rightarrow \mu_2'$ and $(\mu_1, \mu_2') \in S^\sharp$;
        \item if $l$ is external, there exist $\mu_2', \mu_2'', \mu_2'''$ with $\mu_2 \Rightarrow \mu_2' \xrightarrow{l} \mu_2'' \Rightarrow \mu_2'''$ and $(\mu_1, \mu_2''') \in S^\sharp$.
    \end{enumerate}
    We write $\mathcal{P}_1 \preceq_{\mathrm{PFS}} \mathcal{P}_2$ when such a simulation exists.
\end{definition}

\begin{theorem}[Soundness of probabilistic forward simulation]\label{thm:pfs-sound}
    \uses{def:pfs, def:td-preorder}
    %\lean{PLTS.ProbForwardSim.sound}
    %\leanok
    If $\mathcal{P}_1 \preceq_{\mathrm{PFS}} \mathcal{P}_2$ then $\mathcal{P}_1 \sqsubseteq_{\mathrm{TD}} \mathcal{P}_2$.
\end{theorem}

\subsection{Equivalence in the Non-Probabilistic Setting}

When all transitions lead to a Dirac distribution, the PLTS framework collapses to the classical (non-probabilistic) labelled transition system, and the lifted notions admit a more elementary description.

\begin{definition}[Labelled transition system]\label{def:LTS}
    \uses{def:PLTS}
    %\lean{LTS}
    %\leanok
    A PLTS is a \emph{labelled transition system} (LTS) if every transition's target is a Dirac distribution. Equivalently, its transition relation can be viewed as a subset $T \subseteq Q \times L \times Q$.
\end{definition}

\begin{definition}[Weak transitions on an LTS]\label{def:weak-LTS}
    \uses{def:LTS, def:weak-trans}
    %\lean{LTS.Weak}
    %\leanok
    On an LTS, the weak internal transition $\Rightarrow$ coincides with the reflexive transitive closure of internal transitions. The weak external transition is $q \xRightarrow{l} q'$ iff there exist $q_1, q_2$ with $q \Rightarrow q_1 \xrightarrow{l} q_2 \Rightarrow q'$.
\end{definition}

\begin{definition}[Forward simulation]\label{def:fwd-sim}
    \uses{def:weak-LTS}
    %\lean{LTS.ForwardSim}
    %\leanok
    A \emph{forward simulation} of an LTS $\mathcal{L}_1$ by an LTS $\mathcal{L}_2$ is a relation $R \subseteq Q_1 \times Q_2$ with $(q_0^1, q_0^2) \in R$ such that, for every $(q_1, q_2) \in R$ and every $q_1 \xrightarrow{l} q_1'$:
    \begin{enumerate}
        \item if $l$ is internal, there exists $q_2'$ with $q_2 \Rightarrow q_2'$ and $(q_1', q_2') \in R$;
        \item if $l$ is external, there exist $q_2', q_2'', q_2'''$ with $q_2 \Rightarrow q_2' \xrightarrow{l} q_2'' \Rightarrow q_2'''$ and $(q_1', q_2''') \in R$.
    \end{enumerate}
    We write $\mathcal{L}_1 \preceq_{\mathrm{FS}} \mathcal{L}_2$ when such a simulation exists.
\end{definition}

\begin{proposition}[Equivalence on LTS]\label{prop:lts-equiv}
    \uses{def:pfs, def:fwd-sim}
    %\lean{LTS.ForwardSim.equiv_ProbForwardSim}
    %\leanok
    On LTS, forward simulation and probabilistic forward simulation coincide: $\mathcal{L}_1 \preceq_{\mathrm{FS}} \mathcal{L}_2$ iff $\mathcal{L}_1 \preceq_{\mathrm{PFS}} \mathcal{L}_2$.
\end{proposition}

\subsection{Partial-Observation Adversaries}

The scheduler of Definition~\ref{def:scheduler} models an \emph{omniscient} adversary that can inspect the entire execution. To model weaker adversaries one restricts schedulers to depend only on a partial view of the system, defined via \emph{signals}.

\begin{definition}[Partial observation]\label{def:partial-obs}
    \uses{def:PLTS}
    %\lean{PLTS.PartialObservation}
    %\leanok
    A \emph{partial observation} on a PLTS $\mathcal{P}$ is a pair of mappings
    \[
        \mathit{obs}_Q \colon Q \longrightarrow \Sigma_Q, \qquad \mathit{obs}_L \colon T \longrightarrow \Sigma_L,
    \]
    sending states and transitions to their \emph{state signal} and \emph{label signal} in two abstract signal sets $\Sigma_Q, \Sigma_L$.
\end{definition}

\begin{definition}[Consistent scheduler]\label{def:consistent-sched}
    \uses{def:scheduler, def:partial-obs}
    %\lean{PLTS.Scheduler.Consistent}
    %\leanok
    A scheduler $\sigma$ is \emph{consistent} with the partial observation $(\mathit{obs}_Q, \mathit{obs}_L)$ if its output depends on its input only through signals: two finite executions with the same image under the observation produce schedulers that coincide on the corresponding label signals.
\end{definition}

To prevent pathological signal choices that would conflate genuinely distinct non-deterministic alternatives, we restrict to admissible observations.

\begin{definition}[Admissible observation]\label{def:admissible-obs}
    \uses{def:partial-obs}
    %\lean{PLTS.PartialObservation.Admissible}
    %\leanok
    A partial observation is \emph{admissible} when
    \begin{enumerate}
        \item the state signal $\mathit{obs}_Q(q)$ determines the set $\{\mathit{obs}_L(t) \mid t \text{ enabled in } q\}$ of label signals of enabled transitions;
        \item in any state $q$, no two distinct enabled transitions share the same label signal.
    \end{enumerate}
\end{definition}

Under these conditions the adversary always knows which non-deterministic choices are available; partial information arises solely from the probabilistic outcomes of transitions.

\begin{definition}[Belief PLTS]\label{def:belief-PLTS}
    \uses{def:PLTS, def:admissible-obs}
    %\lean{PLTS.Belief}
    %\leanok
    Let $\mathcal{P} = (Q, L, T, q_0)$ be a PLTS equipped with an admissible partial observation. Its \emph{belief PLTS} $\mathcal{B}(\mathcal{P})$ is the PLTS whose
    \begin{itemize}
        \item states are distributions over sets of states sharing a common state signal;
        \item labels are the label signals $\Sigma_L$;
        \item transitions describe the update of the current belief when an underlying transition fires;
        \item initial state is the Dirac on $q_0$.
    \end{itemize}
\end{definition}

\begin{theorem}[Belief reduction]\label{thm:belief-reduction}
    \uses{def:belief-PLTS, def:consistent-sched, def:trace-distribution}
    %\lean{PLTS.Belief.equiv}
    %\leanok
    The trace distributions reachable on $\mathcal{P}$ by schedulers consistent with the partial observation coincide with the trace distributions reachable on $\mathcal{B}(\mathcal{P})$ by arbitrary (omniscient) schedulers.
\end{theorem}

Theorem~\ref{thm:belief-reduction} is the practical lever behind our use of partial information: the soundness of probabilistic forward simulation (Theorem~\ref{thm:pfs-sound}) is stated for omniscient schedulers, and the belief construction lets us reuse it in the partial-information setting.

\subsection{Parallel Composition}

\begin{definition}[CSP-style parallel composition]\label{def:par-composition}
    \uses{def:PLTS}
    %\lean{PLTS.Par}
    %\leanok
    The \emph{parallel composition} $\mathcal{P}_1 \parallel \mathcal{P}_2$ is the CSP-style synchronous product of $\mathcal{P}_1$ and $\mathcal{P}_2$: the two components synchronise on the external labels they share and interleave on all other transitions.
\end{definition}

\begin{theorem}[Pre-congruence]\label{thm:pfs-precongruence}
    \uses{def:pfs, def:par-composition}
    %\lean{PLTS.ProbForwardSim.par_congr}
    %\leanok
    Probabilistic forward simulation is a pre-congruence for parallel composition: if $\mathcal{P}_1 \preceq_{\mathrm{PFS}} \mathcal{P}_1'$ then $\mathcal{P}_1 \parallel \mathcal{P}_2 \preceq_{\mathrm{PFS}} \mathcal{P}_1' \parallel \mathcal{P}_2$ for every $\mathcal{P}_2$. Moreover, partial observations on the components extend canonically to the composition.
\end{theorem}

\begin{proposition}[Failure of compositionality of $\sqsubseteq_{\mathrm{TD}}$]\label{prop:td-not-precongruence}
    \uses{def:td-preorder, def:par-composition}
    %\lean{PLTS.TDPreorder.not_par_congr}
    %\leanok
    The trace distribution pre-order is \emph{not} a pre-congruence for parallel composition: there exist PLTS $\mathcal{P}_1, \mathcal{P}_1', \mathcal{P}_2$ with $\mathcal{P}_1 \sqsubseteq_{\mathrm{TD}} \mathcal{P}_1'$ such that $\mathcal{P}_1 \parallel \mathcal{P}_2 \not\sqsubseteq_{\mathrm{TD}} \mathcal{P}_1' \parallel \mathcal{P}_2$.
\end{proposition}

\begin{theorem}[Largest pre-congruence inside $\sqsubseteq_{\mathrm{TD}}$]\label{thm:pfs-largest-precongruence}
    \uses{thm:pfs-sound, thm:pfs-precongruence, prop:td-not-precongruence}
    %\lean{PLTS.ProbForwardSim.largest_precongr}
    %\leanok
    Probabilistic forward simulation characterises the largest pre-congruence (for CSP-style parallel composition) contained in the trace distribution pre-order. In particular, the simulation pre-order $\preceq_{\mathrm{PFS}}$ is contained in $\sqsubseteq_{\mathrm{TD}}$, and every pre-congruence contained in $\sqsubseteq_{\mathrm{TD}}$ is contained in $\preceq_{\mathrm{PFS}}$.
\end{theorem}
